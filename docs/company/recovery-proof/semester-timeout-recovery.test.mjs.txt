import assert from 'node:assert/strict';
import {recover} from './semester-timeout-recovery.mjs';
const p={maxQueries:3};
const s={state:'executing',commandId:'c1',attemptId:'a1',operationKey:'stable-key',version:1,queries:0};
const ev=(type,x={})=>({type,commandId:'c1',attemptId:'a1',...x});
let r=recover(s,ev('TIMEOUT'),p); assert.equal(r.state.state,'outcome_unknown');
assert.ok(!r.effects.some(x=>x.type==='SEND_WRITE'));
r=recover(r.state,ev('CHECK_DUE'),p); assert.equal(r.state.state,'reconciling'); assert.equal(r.state.queries,1);
let ok=recover(r.state,ev('STATUS',{trusted:true,matchesRequest:true,result:'APPLIED',sourceRef:'sis-result'}),p);
assert.equal(ok.state.state,'confirmed');
assert.equal(recover(ok.state,ev('TIMEOUT'),p).state.version,ok.state.version);
assert.equal(recover(r.state,ev('STATUS',{trusted:true,matchesRequest:true,result:'NOT_FOUND'}),p).state.state,'outcome_unknown');
assert.equal(recover(r.state,ev('STATUS',{trusted:false,matchesRequest:true,result:'APPLIED'}),p).state.state,'manual_review');
let absence={trusted:true,matchesRequest:true,result:'NOT_APPLIED',terminal:true,noEffects:true,originalCannotComplete:true,currentAuthority:true,approvalValid:true,preconditionsValid:true,killSwitch:false};
assert.equal(recover(r.state,ev('STATUS',absence),p).state.state,'ready');
assert.equal(recover(r.state,ev('STATUS',{...absence,currentAuthority:false}),p).state.state,'manual_review');
assert.equal(recover(r.state,ev('STATUS',{...absence,originalCannotComplete:false}),p).state.state,'outcome_unknown');
assert.equal(recover({...r.state,queries:3},ev('LOOKUP_FAILED'),p).state.state,'manual_review');
assert.deepEqual(recover(r.state,ev('STATUS',{...absence,attemptId:'old'}),p).effects,[]);
assert.equal(recover(r.state,ev('STATUS',{trusted:true,matchesRequest:true,result:'REJECTED',terminal:true,noEffects:true}),p).state.state,'failed');
console.log('12 recovery assertions passed');
