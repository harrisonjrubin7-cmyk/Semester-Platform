export function numeric(v,positive=false){if(v===null||v===undefined||typeof v==='boolean'||String(v).trim()==='')throw Error('Value required');const n=Number(v);if(!Number.isFinite(n)||n<0||(positive&&n===0))throw Error('Invalid number');return n;}
export function toECTS(n,unit,ratio=2,aus=null){n=numeric(n);const f=unit==='us'?numeric(ratio,true):unit==='ects'?1:unit==='cats'?.5:unit==='aus'?numeric(aus,true):null;if(f===null)throw Error('Unknown unit');const result=n*f;if(!Number.isFinite(result))throw Error('Overflow');return result;}
export function degree(target,completed,abroadECTS,load,ratio=2){const remaining=Math.max(0,numeric(target,true)-numeric(completed));load=numeric(load,true);const pendingUS=toECTS(abroadECTS,'ects')/numeric(ratio,true);const conditional=Math.max(0,remaining-pendingUS);return {remaining,baselineTerms:Math.ceil(remaining/load),pendingUS,conditional,subsequentTerms:Math.ceil(conditional/load)};}
