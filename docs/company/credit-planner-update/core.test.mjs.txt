import test from 'node:test';import assert from 'node:assert/strict';import {toECTS,degree} from '../core.mjs';import {overlapDays,intervalDays} from '../calendar.mjs';
test('US to ECTS',()=>assert.equal(toECTS(15,'us'),30));test('CATS to ECTS',()=>assert.equal(toECTS(60,'cats'),30));test('Australian explicit factor',()=>assert.equal(toECTS(50,'aus',2,.6),30));test('Australian no default',()=>assert.throws(()=>toECTS(50,'aus')));test('120/60 baseline',()=>assert.equal(degree(120,60,30,15).baselineTerms,4));test('Pending remains separate',()=>assert.equal(degree(120,60,30,15).remaining,60));test('Conditional scenario',()=>assert.equal(degree(120,60,30,15).subsequentTerms,3));for(let v of [-1,'',null,Infinity,true])test('Reject '+String(v),()=>assert.throws(()=>toECTS(v,'us')));
test('Adjacent disjoint intervals',()=>assert.equal(overlapDays('2027-03-01','2027-03-19','2027-03-20','2027-03-28'),0));test('Shared endpoint counts one day',()=>assert.equal(overlapDays('2027-03-01','2027-03-20','2027-03-20','2027-03-28'),1));test('Contained break counts nine',()=>assert.equal(overlapDays('2027-03-01','2027-03-31','2027-03-20','2027-03-28'),9));test('Winter break counts sixteen',()=>assert.equal(intervalDays('2026-12-19','2027-01-03'),16));test('Leap-day count',()=>assert.equal(intervalDays('2028-02-28','2028-03-01'),3));test('Invalid leap date rejected',()=>assert.throws(()=>intervalDays('2027-02-29','2027-03-01')));test('Reversed dates rejected',()=>assert.throws(()=>intervalDays('2027-03-02','2027-03-01')));test('Missing date rejected',()=>assert.throws(()=>intervalDays(null,'2027-03-01')));
